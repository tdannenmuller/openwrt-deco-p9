# À AJOUTER dans /etc/config/network, APRÈS avoir créé le VLAN 832 dans
# LuCI > Réseau > Commutateur (port CPU + port WAN choisi, tous les deux "tagués").
# Ne touche pas aux sections 'lan' / 'wifi-iface' existantes.
#
# À REMPLACER avant usage :
#   - 11:22:33:AA:BB:CC  -> l'adresse MAC de LA VRAIE Livebox remplacée (pas celle du Deco)
#   - Livebox4            -> le modèle réel si ce n'est pas une Livebox 4 (voir README.md)
#   - 0111223344... (clientid/sendopts 61) -> même MAC que macaddr, sans les ':', précédée de 01

# VÉRIFIÉ sur le fil (capture tcpdump sur un Deco, sans ONT, 30/10/2026) : ce bloc produit un DHCPDISCOVER avec nom d'hôte,
# classe fournisseur « sagem » (option 60), identifiant client (61), classe utilisateur (77) et option 90, en VLAN 832.
# NON vérifié : l'acceptation par Orange, la validité du jeton, la priorité 802.1p sur le fil du port WAN.
# Les options « vendorclass » et « userclass » du gabarit d'origine n'existent pas dans le client DHCP d'OpenWrt (ignorées) ;
# « sendopts » 12, 60 et 61 faisaient doublon avec hostname, vendorid et clientid : retirés.

# Marquage de priorité 802.1p (point 4 du README) : sans « type/ifname/vid » la section est ignorée (vérifié : p 0 sur le fil) ;
# complète, les trames DHCP sortent en « vlan 832, p 6 » (vérifié côté processeur ; port du commutateur non vérifié).
config device
	option name 'eth0.832'
	option type '8021q'
	option ifname 'eth0'
	option vid '832'
	list egress_qos_mapping '0:6'
	list egress_qos_mapping '1:0'
	list egress_qos_mapping '6:6'

config interface 'wan'
	option proto 'dhcp'
	option device 'eth0.832'
	option auto '1'
	option hostname 'livebox'
	option vendorid 'sagem'
	option macaddr '11:22:33:AA:BB:CC'
	option clientid '0111223344...'
	option defaultreqopts '0'
	option reqopts '1 3 6 15 28 51 58 59 90 119 120 125'
	list sendopts '77:2b46535644534c5f6c697665626f782e496e7465726e65742e736f66746174686f6d652e4c697665626f7834'
	list sendopts '90:XAUTH'

# NON TESTÉ (IPv6) : le fil lafibre.info dit que le clientid DHCPv6 doit être « 00030001 » + la MAC de la Livebox sans « : »
# (comme l'IPv4), pas la valeur d'exemple ci-dessous.
config interface 'wan6'
	option auto '0'
	option proto 'dhcpv6'
	option device 'eth0.832'
	option reqprefix 'auto'
	option reqaddress 'none'
	option defaultreqopts '0'
	option reqopts '11 17 23 24'
	option noclientfqdn '1'
	option noacceptreconfig '1'
	option clientid '00030001A23456781926'
	option iface_dslite '0'
	list sendopts '11:XAUTH'
	list sendopts '15:FSVDSL_livebox.Internet.softathome.Livebox4'
	list sendopts '16:0000040e0005736167656d'
	list sendopts '17:000005580006000e495056365f524551554553544544'
